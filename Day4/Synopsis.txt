// =======================
// CALLBACK
// =======================

function getData(cb) {
  setTimeout(() => {
    console.log("Data fetched");
    cb();
  }, 1000);
}

getData(() => console.log("Next step"));

  let ok = true;
  ok ? res("Success") : rej("Error");
});

p.then(console.log).catch(console.error);
